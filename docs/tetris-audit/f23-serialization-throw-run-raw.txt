const SAVED_SESSION_KEY = 'tetris.savedGameSession';
const IS_SESSION_SAVED_KEY = 'tetris.isSessionSaved';

// One-shot: the next session-shaped stringify throws, then restores itself.
function armStringify() {
  const original = JSON.stringify;
  JSON.stringify = function (value, ...rest) {
    if (value && typeof value === 'object' && 'gameBoard' in value && 'nextTetrominos' in value) {
      JSON.stringify = original;
      throw new Error('injected serialization failure');
    }
    return original.call(this, value, ...rest);
  };
  return original;
}

function setup() {
  const base = createMemoryStorage();
  const writes = [];
  const storage = {
    getItem: k => base.getItem(k),
    setItem: (k, v) => { writes.push([k, String(v)]); base.setItem(k, v); },
    removeItem: k => base.removeItem(k),
  };
  const scheduler = createFakeScheduler();
  const game = new GameManager({ scheduler, storage });
  let faults = 0, changes = 0;
  game.onFault = () => { faults++; };
  game.onChange = () => { changes++; };
  const origCheck = game.findInvariantViolation.bind(game);
  let checks = 0;
  game.findInvariantViolation = () => { checks++; return origCheck(); };
  const origFailSafe = game.failSafe.bind(game);
  let failSafeCalls = 0;
  game.failSafe = (...a) => { failSafeCalls++; return origFailSafe(...a); };
  return { game, storage, scheduler, writes, counts: () => ({ faults, changes, checks, failSafeCalls }) };
}

test('scratch F23: a serialization throw during the pause save is swallowed by saveGameSession', () => {
  const { game, storage, scheduler, writes, counts } = setup();
  game.handleAction(PlayerAction.newGame);
  assert.equal(game.state, GameState.playing);

  // Control: a real pause writes the save text and the flag.
  game.handleAction(PlayerAction.pause);
  const savedText = storage.getItem(SAVED_SESSION_KEY);
  assert.equal(typeof savedText, 'string', 'control pause wrote a save');
  assert.equal(game.isSessionSaved, true);
  const writesAfterControl = writes.length;
  game.handleAction(PlayerAction.resume);
  assert.equal(game.state, GameState.playing);

  const original = armStringify();
  const checksBefore = counts().checks;
  const pendingBefore = scheduler.pending;
  let escaped = null;
  try {
    game.handleAction(PlayerAction.pause);
  } catch (e) { escaped = e; }
  const pendingAfter = scheduler.pending;

  assert.equal(escaped, null, 'nothing escapes the guarded pause action');
  assert.equal(JSON.stringify, original, 'the one-shot patch restored itself');
  assert.equal(game.state, GameState.paused, 'the pause itself completed');
  assert.equal(counts().faults, 0, 'the onFault callback is not called');
  assert.equal(counts().failSafeCalls, 0, 'failSafe is not called (spied)');
  assert.equal(game.faults.length, 0, 'the faults list stays empty');
  assert.equal(pendingBefore, 1, 'one timer pending while playing (gravity)');
  assert.equal(pendingAfter, 0, 'zero pending after the pause: the pause stopped gravity before the save ran (:494-499)');
  assert.ok(counts().checks > checksBefore, 'the invariant boundary ran after the swallowed failure');
  assert.equal(storage.getItem(SAVED_SESSION_KEY), savedText, 'the previous save text is preserved byte-for-byte (setItem was never reached)');
  assert.ok(!writes.slice(writesAfterControl).some(([k]) => k === SAVED_SESSION_KEY), 'no write to the save key during the failed save');
  assert.equal(game.isSessionSaved, false, 'the catch marks the session not saved');
  assert.equal(storage.getItem(IS_SESSION_SAVED_KEY), 'false', 'the persisted flag now reads false');
});

test('scratch F23: direct saveGameSession call - swallowed throw, later save recovers', () => {
  const { game, storage, scheduler, writes, counts } = setup();
  game.handleAction(PlayerAction.newGame);

  const before = storage.getItem(SAVED_SESSION_KEY);
  const writesBefore = writes.length;
  armStringify();
  let escaped = null;
  let ret = 'sentinel';
  try {
    ret = game.saveGameSession();
  } catch (e) { escaped = e; }
  assert.equal(escaped, null, 'no throw escapes the direct call');
  assert.equal(ret, undefined, 'the direct call returns undefined (captured)');
  assert.equal(storage.getItem(SAVED_SESSION_KEY), before, 'storage contents unchanged by the failed save');
  assert.ok(!writes.slice(writesBefore).some(([k]) => k === SAVED_SESSION_KEY), 'setItem not reached');
  assert.equal(counts().faults, 0, 'no fault recorded');
  assert.equal(counts().failSafeCalls, 0, 'failSafe not called (spied)');
  assert.equal(game.faults.length, 0, 'faults list empty');
  assert.equal(game.isSessionSaved, false, 'flag cleared by the catch');
  assert.ok(writes.slice(writesBefore).some(([k, v]) => k === IS_SESSION_SAVED_KEY && v === 'false'), "the flag key is written 'false' through the setter");

  // A later ordinary save succeeds.
  game.saveGameSession();
  assert.equal(game.isSessionSaved, true, 'a later save succeeds');
  const text = storage.getItem(SAVED_SESSION_KEY);
  assert.doesNotThrow(() => JSON.parse(text), 'the new save text parses');
});
